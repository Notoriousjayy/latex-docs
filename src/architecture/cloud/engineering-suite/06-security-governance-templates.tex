\section{How to apply the forms}
Use the source's separation of threats (Chapter 7), access controls (Chapter 10), data controls (Chapter 11) and cloud management (Chapter 12). A template is complete only when the control owner and a negative-case test are identified.
\clearpage\section{S01. Threat and trust-boundary record}
\sourceRef{7}{159}\\[8pt]
\field{Asset, use case and business importance}
\field{Threat agent and attack surface}
\field{Organizational trust boundary}
\field{Network or identity trust boundary}
\field{Threat scenario and preconditions}
\field{Impact and existing mitigating controls}
\field{Likelihood / impact rationale}
\field{Residual risk owner and acceptance}
\begin{workbox}[title={Verification gate}]
\begin{itemize}
\item Do not treat network reachability as authorization.
\item Record who can cross each trust boundary.
\end{itemize}
\end{workbox}
\clearpage\section{S02. IAM and authentication control plan}
\sourceRef{10}{269}\\[8pt]
\field{Identity source and federation assumptions}
\field{Human and workload identities}
\field{Role / permission model and least-privilege limits}
\field{MFA factor and recovery process}
\field{SSO lifetime and session revocation}
\field{Privileged break-glass process}
\field{Joiner / mover / leaver workflow}
\field{Audit event and access review evidence}
\begin{workbox}[title={Verification gate}]
\begin{itemize}
\item Test an explicitly unauthorized action.
\item Demonstrate deprovisioned identity loses access.
\end{itemize}
\end{workbox}
\clearpage\section{S03. Network access and trust-zone matrix}
\sourceRef{8}{195}\\[8pt]
\field{Source and destination identity or subnet}
\field{Protocol, direction and port}
\field{Business justification for allowed flow}
\field{Security group / firewall / VPN policy owner}
\field{Ingress, egress and lateral segmentation}
\field{Monitoring source and flow log location}
\field{Negative connection test and disposition}
\begin{workbox}[title={Verification gate}]
\begin{itemize}
\item Default-deny intent is explicit.
\item Routes, gateways and policy enforcement are consistent.
\end{itemize}
\end{workbox}
\clearpage\section{S04. Data classification and protection plan}
\sourceRef{11}{311}\\[8pt]
\field{Data category and regulated characteristics}
\field{Data owner and retention requirements}
\field{Locations including replicas, caches and backups}
\field{Encryption in transit and at rest, key owner}
\field{DLP policy and allowed exception path}
\field{Recovery objectives and restoration tests}
\field{Data deletion / export / exit requirements}
\begin{workbox}[title={Verification gate}]
\begin{itemize}
\item Check that copies follow original data governance.
\item Protect and test backups separately from live data.
\end{itemize}
\end{workbox}
\clearpage\section{S05. Container and workload hardening checklist}
\sourceRef{6}{115}\\[8pt]
\field{Image registry and immutable reference}
\field{Software components and provenance evidence}
\field{Base image patch and rebuild policy}
\field{Runtime user, permissions and filesystem restrictions}
\field{Pod / host / overlay networking and policy}
\field{Secrets injection and rotation design}
\field{Log and artifact retention}
\field{Runtime isolation and escape assumptions}
\begin{workbox}[title={Verification gate}]
\begin{itemize}
\item Use admission/baseline enforcement where available.
\item Test privileged execution and outbound egress denials.
\end{itemize}
\end{workbox}
\clearpage\section{S06. Security logging and detection coverage}
\sourceRef{10}{269}\\[8pt]
\field{Threat / event scenario}
\field{Security control and telemetry producer}
\field{Event schema and identity attribution}
\field{Collection, retention and immutable storage}
\field{Alert threshold and noise tuning}
\field{Triage owner and escalation timeline}
\field{Replay procedure and observed alert}
\field{Evidence record and closure outcome}
\begin{workbox}[title={Verification gate}]
\begin{itemize}
\item Simulate an event end to end.
\item Verify silent failures and ingestion gaps are observable.
\end{itemize}
\end{workbox}
\clearpage\section{S07. Control verification and exception log}
\sourceRef{12}{325}\\[8pt]
\field{Control identifier and requirement}
\field{Accountable team and operational authority}
\field{Configuration source and applied version}
\field{Test preconditions / positive path}
\field{Negative test / expected denial}
\field{Evidence URI and reviewer}
\field{Exception, expiration and residual risk}
\field{Remediation and retest closure}
\begin{workbox}[title={Verification gate}]
\begin{itemize}
\item Each exception has an end date.
\item Closure demands a new test rather than a comment.
\end{itemize}
\end{workbox}
\clearpage\section{S08. Incident and recovery readiness}
\sourceRef{11}{311}\\[8pt]
\field{Triggering signals and affected assets}
\field{Severity and decision authority}
\field{Containment without destroying evidence}
\field{Communication and contractual obligations}
\field{Recovery mechanism and restore order}
\field{Observed elapsed interruption and data loss}
\field{Root-cause and prevention owner}
\field{Verification and postincident review}
\begin{workbox}[title={Verification gate}]
\begin{itemize}
\item Preserve access to audit records during outages.
\item Recovery exercise measures actual restoration.
\end{itemize}
\end{workbox}
