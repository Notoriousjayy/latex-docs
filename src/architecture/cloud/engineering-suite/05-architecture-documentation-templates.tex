\section{Usage and conventions}
Copy the relevant blank template into a project architecture description or document a completed form in a versioned repository. Fields requiring proof should link to diagrams, test results or accountable review decisions. Template structure is original; headings reference the book's roles, mechanisms and architectures.
\clearpage\section{T01. System context and stakeholders}
\sourceRef{4}{51}\\[8pt]
\field{System and service identifier}
\field{Business objective and critical capabilities}
\field{Cloud provider, consumer and broker}
\field{Service owner and administrators}
\field{Primary workloads and peak traffic}
\field{Systems and datasets outside the boundary}
\field{Constraints, assumptions and unknowns}
\begin{workbox}[title={Review gate}]
\begin{itemize}
\item Context diagram identifies parties and trust crossings.
\item Out-of-scope systems and compliance responsibilities are explicit.
\end{itemize}
\end{workbox}
\clearpage\section{T02. Delivery and deployment model decision}
\sourceRef{4}{51}\\[8pt]
\field{IaaS / PaaS / SaaS candidate and rationale}
\field{Public / private / hybrid / multicloud placement}
\field{Provider-managed layers}
\field{Consumer-managed layers}
\field{Alternatives and portability consequences}
\field{Approval authority and expiration date}
\begin{workbox}[title={Review gate}]
\begin{itemize}
\item Responsibilities vary by model and are reviewed with the provider.
\item Every retained consumer control has a named operator.
\end{itemize}
\end{workbox}
\clearpage\section{T03. Functional and quality-attribute scenarios}
\sourceRef{3}{23}\\[8pt]
\field{Business process or use case}
\field{Actor and stimulus}
\field{Environment and workload intensity}
\field{Expected response and success threshold}
\field{Failure / degraded scenario}
\field{Evidence owner and test location}
\begin{workbox}[title={Review gate}]
\begin{itemize}
\item Scenario is measurable.
\item Failure and normal paths are both specified.
\end{itemize}
\end{workbox}
\clearpage\section{T04. Mechanism composition record}
\sourceRef{8}{195}\\[8pt]
\field{Required capability}
\field{Source mechanism names and page references}
\field{Resource prerequisites}
\field{Control-plane dependencies}
\field{Data-flow and network paths}
\field{Observability and metering signals}
\field{Positive and negative test cases}
\begin{workbox}[title={Review gate}]
\begin{itemize}
\item Controls are owned and logged.
\item A mechanism is not used as a substitute for a full system design.
\end{itemize}
\end{workbox}
\clearpage\section{T05. Architecture decision record (ADR)}
\sourceRef{13}{343}\\[8pt]
\field{ADR identifier, status and date}
\field{Context / problem statement}
\field{Selected architecture name and source location}
\field{Alternatives and why rejected}
\field{Mechanism prerequisites and dependencies}
\field{Security and resilience tradeoffs}
\field{Financial and operating implications}
\field{Acceptance measures and revisit trigger}
\begin{workbox}[title={Review gate}]
\begin{itemize}
\item Decision explains why complexity is justified.
\item Test evidence can falsify the decision.
\end{itemize}
\end{workbox}
\clearpage\section{T06. Availability and disaster recovery}
\sourceRef{14}{371}\\[8pt]
\field{Critical business service and dependencies}
\field{Fault domains and shared dependencies}
\field{Maximum acceptable interruption (RTO)}
\field{Maximum acceptable data loss (RPO)}
\field{Failover sequence and rollback}
\field{Backup location, retention and restoration owner}
\field{Injectable failure scenario and result}
\begin{workbox}[title={Review gate}]
\begin{itemize}
\item Recovery is tested, not inferred from replication.
\item Dependence on common control-plane services is analyzed.
\end{itemize}
\end{workbox}
\clearpage\section{T07. Architecture review checklist}
\sourceRef{16}{459}\\[8pt]
\field{Evidence package and review date}
\field{Consumer/provider control ownership}
\field{Security threat model and exceptions}
\field{Network and data-location constraints}
\field{Deployment and rollback approach}
\field{Monitoring, SLA and billing evidence}
\field{Exit and portability review}
\begin{workbox}[title={Review gate}]
\begin{itemize}
\item Unresolved high-impact risks have an accountable approver.
\item Operational handoff is explicit.
\end{itemize}
\end{workbox}
\clearpage\section{T08. Architecture traceability matrix}
\sourceRef{15}{415}\\[8pt]
\field{Requirement identifier and rationale}
\field{Selected mechanisms}
\field{Selected architecture(s)}
\field{Control owner / party}
\field{Validation procedure and outcome}
\field{Defect or exception reference}
\field{Reevaluation trigger}
\begin{workbox}[title={Review gate}]

\item Unsupported claims are marked as assumptions.
\end{itemize}
\end{workbox}
