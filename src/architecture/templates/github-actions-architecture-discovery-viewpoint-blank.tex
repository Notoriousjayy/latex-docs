\section{Evidence class and promotion}
\Field{Observed (E1) definition}{Require immutable GitHub configuration key or file-path evidence.}
\Field{Inferred (E2) definition}{Capture rule version, matched paths, ambiguity and uncertainty.}
\Field{Verified (E3) definition}{Require independently corroborating registry, deployment, service-catalog or approved owner records.}
\Field{Nonpromotable claims and exceptions}{Record uncertainty, failed lookups and negative results without silently upgrading confidence.}

\section{Model construction and inspection}
\Field{Construction method}{Specify reproducible revision pinning and parse rules.}
\Field{Interpretation method}{Explain evidence annotations, legends and prohibited inferences.}
\Field{Correspondence method}{Define how IDs and links reconcile across repository, job, component and deployment views.}
\Field{Analysis method}{Document consistency, coverage, missing data, impact analysis and drift detection.}

\section{Conformance, quality and update}
\Field{Model validation}{Define deterministic rebuild, schema validation and counterexamples.}
\Field{Viewpoint review}{Identify reviewer and acceptance criteria.}
\Field{Versioning and revision rule}{Track changes to model grammar, evidence semantics and status labels.}
\section{Record review and sign-off}
\Field{Prepared by, role, and revision}{Provide author, contact, version, and creation date.}
\Field{Reviewed by, date, and scope}{Record who reviewed which evidence and what remains open.}
\Field{Status and next action}{Distinguish unreviewed, draft, reviewed, approved, and superseded; attach approvals separately.}
