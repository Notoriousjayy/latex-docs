cmake_minimum_required(VERSION 3.18)
project(animation_timing_suite LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(animation_timing INTERFACE)
target_include_directories(animation_timing INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(animation_timing_demo examples/timing_demo.cpp)
target_link_libraries(animation_timing_demo PRIVATE animation_timing)
add_executable(animation_timing_tests tests/test_animation.cpp)
target_link_libraries(animation_timing_tests PRIVATE animation_timing)
enable_testing()
add_test(NAME animation_timing_tests COMMAND animation_timing_tests)
