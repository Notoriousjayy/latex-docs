cmake_minimum_required(VERSION 3.16)
project(compiler_suite_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(executable compiler_dataflow compiler_scheduler compiler_allocator)
  if(executable STREQUAL "compiler_dataflow")
    set(source dataflow.cpp)
  elseif(executable STREQUAL "compiler_scheduler")
    set(source scheduler.cpp)
  else()
    set(source allocator.cpp)
  endif()
  add_executable(${executable} ${source})
  target_compile_options(${executable} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
enable_testing()
add_test(NAME dataflow COMMAND compiler_dataflow)
add_test(NAME scheduler COMMAND compiler_scheduler)
add_test(NAME allocator COMMAND compiler_allocator)
