cmake_minimum_required(VERSION 3.16)
project(parsing_suite LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(parsing_cyk cyk.cpp)
target_include_directories(parsing_cyk PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(parsing_cyk PRIVATE -Wall -Wextra -Wpedantic)
add_executable(test_cyk tests/test_cyk.cpp)
target_link_libraries(test_cyk PRIVATE parsing_cyk)
enable_testing()
add_test(NAME cyk_reference COMMAND test_cyk)
add_library(parsing_earley earley.cpp)
target_include_directories(parsing_earley PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(parsing_earley PRIVATE -Wall -Wextra -Wpedantic)
add_executable(test_earley tests/test_earley.cpp)
target_link_libraries(test_earley PRIVATE parsing_earley)
add_test(NAME earley_reference COMMAND test_earley)
