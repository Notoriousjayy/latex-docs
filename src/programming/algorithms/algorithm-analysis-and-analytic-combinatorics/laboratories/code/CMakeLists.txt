cmake_minimum_required(VERSION 3.16)
project(aaoa_laboratories LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_executable(analysis_cpp instrumented_analysis.cpp)
add_executable(analysis_cpp_tests test_instrumented.cpp)
add_executable(occupancy_c occupancy_model.c)
enable_testing()
add_test(NAME analysis_cpp_invariants COMMAND analysis_cpp_tests)
add_test(NAME occupancy_c_invariants COMMAND occupancy_c 64 8 71)
