cmake_minimum_required(VERSION 3.16)
project(algorithm_engineering_labs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(algorithm_headers INTERFACE)
target_include_directories(algorithm_headers INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(algorithm_headers INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(alg_tests tests/test_algorithms.cpp)
target_link_libraries(alg_tests PRIVATE algorithm_headers)
add_executable(alg_bench src/benchmark.cpp)
target_link_libraries(alg_bench PRIVATE algorithm_headers)
enable_testing()
add_test(NAME algorithms COMMAND alg_tests)
