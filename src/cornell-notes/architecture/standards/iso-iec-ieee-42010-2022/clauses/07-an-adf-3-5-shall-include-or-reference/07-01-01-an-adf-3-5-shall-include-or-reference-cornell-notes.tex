\begin{CornellOverviewBox}{Overview}
{\Large\bfseries\textcolor{CNavy}{Section 7.1.1: An ADF (3.5) shall include or reference}}\\[3pt]
{\small\textbf{Classification:} Normative}\\
{\small\textbf{Learning objective:} Identify the mandatory, recommended, and permitted provisions governing an adf (3.5) shall include or reference.}
\end{CornellOverviewBox}
\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Purpose}{Defines the information an architecture description framework must include or reference.}
\CornellNoteRow{Requirement}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item An ADF (3.5) must include or reference:
\item Required content: information identifying the ADF and its intended scope of applicability
\item Required content: version identification of the ADF as specified by the organization and/or project
\item Required content: one or more typical stakeholders (per 6.2)
\item Required content: one or more typical concerns held by typical stakeholders (per 6.4)
\item Required content: one or more architecture viewpoints that frame those typical concerns (per 8.1)
\end{itemize}}
\CornellNoteRow{Related sections}{6.2, 6.4, 8.1}
\CornellNoteRow{Conformance check}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Required content: information identifying the ADF and its intended scope of applicability
\item Required content: version identification of the ADF as specified by the organization and/or project
\end{itemize}}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Which provisions in An ADF (3.5) shall include or reference are mandatory for conformance?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
Defines the information an architecture description framework must include or reference. For conformance, its mandatory provisions must be satisfied, including: Required content: information identifying the ADF and its intended scope of applicability; Required content: version identification of the ADF as specified by the organization and/or project.
\end{CornellSummaryBox}
