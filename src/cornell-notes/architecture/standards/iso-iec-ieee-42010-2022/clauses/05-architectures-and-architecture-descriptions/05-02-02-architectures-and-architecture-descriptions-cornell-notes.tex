\begin{CornellOverviewBox}{Overview}
{\Large\bfseries\textcolor{CNavy}{Section 5.2.2: Architectures and architecture descriptions}}\\[3pt]
{\small\textbf{Classification:} Conceptual foundation}\\
{\small\textbf{Learning objective:} Explain the role of architectures and architecture descriptions and connect it to related architecture-description concepts.}
\end{CornellOverviewBox}
\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Purpose / key concept}{The architecture of an entity of interest comprises the fundamental concepts or properties of that entity considered in its environment.}
\CornellNoteRow{Core details}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item An architecture description is an expression of an architecture.
\item Architecture descriptions are work products resulting from architecting efforts.
\item As a work product, an AD is devised for the specific purpose for which the architecting effort is undertaken, which is distinct from the purpose of the entity of interest.
\item An AD comprises AD elements (see 5.2.10).
\end{itemize}}
\CornellNoteRow{Distinction}{Architecture is abstract; an architecture description is the tangible work product that expresses it.}
\CornellNoteRow{Interpretive note}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item ISO/IEC/IEEE 42020 specifies a set of processes for architecting which can be employed in support of creating one or more ADs.
\end{itemize}}
\CornellNoteRow{Related sections}{5.2.10, 5.2.3, 5.2.4}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item What is the central role of Architectures and architecture descriptions?
\item How does Architectures and architecture descriptions connect to adjacent architecture-description concepts?
\item What closely related concepts must be kept distinct?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
The architecture of an entity of interest comprises the fundamental concepts or properties of that entity considered in its environment. Architecture is abstract; an architecture description is the tangible work product that expresses it.
\end{CornellSummaryBox}
