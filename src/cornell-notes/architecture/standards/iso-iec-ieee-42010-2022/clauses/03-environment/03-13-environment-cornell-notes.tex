\begin{CornellNotesTable}
\CornellNoteRow{Term}{\textbf{environment}}
\CornellNoteRow{Definition}{The surrounding context of things, conditions, and influences affecting an entity.}
\CornellNoteRow{Key characteristics}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item The environment of an entity of interest (3.12) includes external entities that can have various influences upon an entity, such as developmental, technological, business, operational, organizational, political, economic, legal, regulatory, ecological and social influences as well as external physical effects such as electromagnetic radiation, charged particles, gravitational effects, and electric and magnetic fields.
\item A label attached as a qualifier to the term environment identifies a particular context within another context, such as development environment, test environment, and operational environment.
\end{itemize}}
