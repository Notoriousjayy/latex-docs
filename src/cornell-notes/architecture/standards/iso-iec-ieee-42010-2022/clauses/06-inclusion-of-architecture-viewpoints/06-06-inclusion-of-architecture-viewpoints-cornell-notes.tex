\begin{CornellOverviewBox}{Overview}
{\Large\bfseries\textcolor{CNavy}{Section 6.6: Inclusion of architecture viewpoints}}\\[3pt]
{\small\textbf{Classification:} Normative}\\
{\small\textbf{Learning objective:} Identify the mandatory, recommended, and permitted provisions governing inclusion of architecture viewpoints.}
\end{CornellOverviewBox}
\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Purpose}{Establishes how viewpoints are included or referenced and how they cover identified concerns and stakeholder perspectives.}
\CornellNoteRow{Requirement}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item A conforming architecture description must include or reference each architecture viewpoint used therein.
\item Each architecture viewpoint must include version identification as specified by the organization and/ or project.
\item The specification of Each included architecture viewpoint must be in accordance with the provisions of Clause 8.
\item Each concern identified in accordance with 6.4 must be framed by at least one architecture viewpoint.
\item Each stakeholder perspective identified in accordance with 6.3 must be associated with the architecture viewpoints which cover that perspective.
\end{itemize}}
\CornellNoteRow{Example / application}{Function, Information, Resource, and Organization.}
\CornellNoteRow{Interpretive note}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item This section does not require the use of any particular architecture viewpoints.
\item Annex B and Annex C provide additional information pertaining to specification of architecture viewpoints.
\end{itemize}}
\CornellNoteRow{Related sections}{6.4, 6.3}
\CornellNoteRow{Conformance check}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item A conforming architecture description must include or reference each architecture viewpoint used therein
\item Each architecture viewpoint must include version identification as specified by the organization and/ or project
\end{itemize}}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Which provisions in Inclusion of architecture viewpoints are mandatory for conformance?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
Establishes how viewpoints are included or referenced and how they cover identified concerns and stakeholder perspectives. For conformance, its mandatory provisions must be satisfied, including: A conforming architecture description must include or reference each architecture viewpoint used therein; Each architecture viewpoint must include version identification as specified by the organization and/ or project.
\end{CornellSummaryBox}
