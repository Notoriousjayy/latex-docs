\begin{CornellOverviewBox}{Overview}
{\Large\bfseries\textcolor{CNavy}{Section 6.9.2: Correspondences}}\\[3pt]
{\small\textbf{Classification:} Normative}\\
{\small\textbf{Learning objective:} Identify the mandatory, recommended, and permitted provisions governing correspondences.}
\end{CornellOverviewBox}
\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Purpose}{Establishes the information an architecture description must record about correspondences among AD elements.}
\CornellNoteRow{Requirement}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item A conforming architecture description must include or reference a list of AD element correspondences.
\item An AD element correspondence must identify one or more participating AD elements.
\item An AD element correspondence must identify any governing correspondence methods (see 6.9.3).
\item Each AD element correspondence must identify the participating ADs.
\end{itemize}}
\CornellNoteRow{Permission}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item An AD element correspondence may involve elements within an AD or across several ADs.
\end{itemize}}
\CornellNoteRow{Example / application}{An AD element satisfies a Requirement as demonstrated by an Evaluation Method: TracesToDemo (an AD Element, a Requirement, an Evaluation Method).}
\CornellNoteRow{Interpretive note}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item AD element Correspondences can be used to express relations among ADs, ADFs, and ADLs. See Zachman Framework example in Annex F.
\end{itemize}}
\CornellNoteRow{Related sections}{6.9.3}
\CornellNoteRow{Conformance check}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item A conforming architecture description must include or reference a list of AD element correspondences
\item An AD element correspondence must identify one or more participating AD elements
\end{itemize}}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Which provisions in Correspondences are mandatory for conformance?
\item Which provisions are recommendations or permissions rather than requirements?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
Establishes the information an architecture description must record about correspondences among AD elements. For conformance, its mandatory provisions must be satisfied, including: A conforming architecture description must include or reference a list of AD element correspondences; An AD element correspondence must identify one or more participating AD elements. The section also explicitly permits: An AD element correspondence may involve elements within an AD or across several ADs.
\end{CornellSummaryBox}
