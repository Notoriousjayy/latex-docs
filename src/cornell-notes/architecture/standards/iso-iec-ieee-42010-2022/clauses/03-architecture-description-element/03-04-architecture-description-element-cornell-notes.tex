\begin{CornellNotesTable}
\CornellNoteRow{Term}{\textbf{architecture description element}}
\CornellNoteRow{Definition}{An identified or named part of an architecture description.}
\CornellNoteRow{Key characteristics}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item AD elements include stakeholders (3.17), concerns (3.10), stakeholder perspectives (3.18), and aspects (3.9) identified in an AD (3.3), ADLs (3.6), ADFs (3.5) and correspondences (3.11) and correspondence methods used in an AD, and architecture views (3.7), view components (3.19), architecture viewpoints (3.8), and model kinds (3.15) included in an AD (3.3).
\item For the purpose of correspondences (3.11), an AD (3.3) can be considered as an AD element in another AD (3.3).
\end{itemize}}
\CornellNoteRow{Distinction}{An AD element is an occurrence or named part within a description; it is not limited to models and can include stakeholders, concerns, viewpoints, views, correspondences, and other constructs.}
\CornellNoteRow{Related sections}{3.3, 3.17, 3.10, 3.18, 3.9, 3.6, 3.5, 3.11}

\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
An identified or named part of an architecture description. An AD element is an occurrence or named part within a description; it is not limited to models and can include stakeholders, concerns, viewpoints, views, correspondences, and other constructs.
\end{CornellSummaryBox}
