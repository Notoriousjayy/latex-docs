\begin{CornellOverviewBox}{Overview}
{\Large\bfseries\textcolor{CNavy}{Section 5.2.7: Architecture views and architecture viewpoints}}\\[3pt]
{\small\textbf{Classification:} Conceptual foundation}\\
{\small\textbf{Learning objective:} Explain the role of architecture views and architecture viewpoints and connect it to related architecture-description concepts.}
\end{CornellOverviewBox}
\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Purpose / key concept}{An AD contains one or more architecture views.}
\CornellNoteRow{Core details}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item An architecture viewpoint governs one or more of these architecture views.
\item A specification of an architecture viewpoint establishes the conventions for creating, interpreting, presenting and analysing a view to address the concerns framed by that viewpoint.
\end{itemize}}
\CornellNoteRow{Distinction}{A viewpoint frames concerns and governs views; a view addresses those concerns using the viewpoint’s conventions.}
\CornellNoteRow{Study relationship / visual insight}{Study relationship: a viewpoint frames concerns and governs architecture views; a view addresses the framed concerns, is contained in an architecture description, and reflects relevant aspects of the entity.}
\CornellNoteRow{Example / application}{A telecommunications network (entity) is represented by a network connectivity deployment diagram that can be used to express the network model (as an architecture view) contained in a telecommunications architecture description. That view addresses communications parameters such as throughput and uptime (concerns) of operators and users (stakeholders).}
\CornellNoteRow{Interpretive note}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Architecture concerns, stakeholder perspectives and aspects can serve as an organizing basis for the viewpoints of an AD. Aspects are refinements of concerns and these aspects can be used to establish the viewpoint conventions. These aspects are evidenced in the resulting architecture view(s).
\end{itemize}}
\CornellNoteRow{Related sections}{5.2.3, 5.2.9}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item What is the central role of Architecture views and architecture viewpoints?
\item How does Architecture views and architecture viewpoints connect to adjacent architecture-description concepts?
\item What closely related concepts must be kept distinct?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
An architecture description contains one or more architecture views. A viewpoint frames concerns and governs one or more views; the resulting views address those concerns and reflect relevant aspects. Viewpoint conventions guide creation, interpretation, presentation, and analysis of the views.
\end{CornellSummaryBox}
