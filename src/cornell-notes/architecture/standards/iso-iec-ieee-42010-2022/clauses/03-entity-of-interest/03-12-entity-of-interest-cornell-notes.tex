\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Term}{\textbf{entity of interest}}
\CornellNoteRow{Definition}{The entity whose architecture is the subject of an architecture description.}
\CornellNoteRow{Key characteristics}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Here, the term entity of interest refers to the entity whose architecture (3.2) is under consideration in the preparation of an architecture description (3.3).
\item A key boundary distinguishes the entity of interest from other entities which are not the subject of the architecture description (3.3).
\item Here, interest in an entity is intended to encompass interest in that entity’s environment, life cycle, architecture, requirements, design, implementation and operation. Such interests are captured via aspects, concerns and stakeholder perspectives.
\end{itemize}}
\CornellNoteRow{Distinction}{The entity of interest is the subject being architected/described; its environment consists of surrounding influences and entities.}
\CornellNoteRow{Example / application}{Enterprise, organization, solution, system (including software systems), subsystem, process, business, data (as a data item or data structure), application, information technology (as a collection), mission, product, service, software item, hardware item, product line, family of systems, system of systems, collection of systems, collection of applications.}
\CornellNoteRow{Related sections}{3.3, 3.2}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item How would you explain entity of interest in your own words?
\item Which neighboring concept is easiest to confuse with entity of interest, and how is it different?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
The entity whose architecture is the subject of an architecture description. The entity of interest is the subject being architected/described; its environment consists of surrounding influences and entities.
\end{CornellSummaryBox}
