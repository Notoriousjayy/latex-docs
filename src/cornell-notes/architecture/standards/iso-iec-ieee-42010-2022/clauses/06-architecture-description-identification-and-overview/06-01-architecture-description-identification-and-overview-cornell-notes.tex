\begin{CornellOverviewBox}{Overview}
{\Large\bfseries\textcolor{CNavy}{Section 6.1: Architecture description identification and overview}}\\[3pt]
{\small\textbf{Classification:} Normative}\\
{\small\textbf{Learning objective:} Identify the mandatory, recommended, and permitted provisions governing architecture description identification and overview.}
\end{CornellOverviewBox}
\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Purpose}{Establishes the minimum identification and overview information that an architecture description must provide.}
\CornellNoteRow{Requirement}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item A conforming architecture description must identify the entity of interest and the expected environment of that entity of interest.
\item A conforming architecture description must include a statement of its intended purpose.
\item A conforming architecture description must include identifying information and supplementary information as determined by the project and/or organization.
\end{itemize}}
\CornellNoteRow{Example / application}{Date of issue and status; authors, reviewers, approving authority, issuing organization; change history; summary; scope; context; glossary; version control information; configuration management information and references. See ISO/IEC/IEEE 15289 or ISO/IEC TS 33060 for additional examples.}
\CornellNoteRow{Interpretive note}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item For an AD intended to serve as a reference for another AD, the entity of interest is abstract, or a generalization of entities of interest, and the purpose of the AD is to express a reference architecture which could provide a basis for further ADs.
\item The scope does not prescribe a format for ADs.
\end{itemize}}
\CornellNoteRow{Conformance check}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item A conforming architecture description must identify the entity of interest and the expected environment of that entity of interest
\item A conforming architecture description must include a statement of its intended purpose
\end{itemize}}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Which provisions in Architecture description identification and overview are mandatory for conformance?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
Establishes the minimum identification and overview information that an architecture description must provide. For conformance, its mandatory provisions must be satisfied, including: A conforming architecture description must identify the entity of interest and the expected environment of that entity of interest; A conforming architecture description must include a statement of its intended purpose.
\end{CornellSummaryBox}
