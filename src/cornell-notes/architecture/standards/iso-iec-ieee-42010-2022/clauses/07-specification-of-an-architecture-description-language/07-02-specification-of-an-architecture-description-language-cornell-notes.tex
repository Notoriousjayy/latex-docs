\begin{CornellOverviewBox}{Overview}
{\Large\bfseries\textcolor{CNavy}{Section 7.2: Specification of an architecture description language}}\\[3pt]
{\small\textbf{Classification:} Normative}\\
{\small\textbf{Learning objective:} Identify the mandatory, recommended, and permitted provisions governing specification of an architecture description language.}
\end{CornellOverviewBox}
\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Purpose}{Defines the minimum specification content for an architecture description language.}
\CornellNoteRow{Requirement}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item The specification of an ADL must include or reference:
\item Required content: identification of typical concerns (6.4) or aspects (6.5) covered by the ADL
\item Required content: the identification of one or more view methods to be selected from the ADL (per 8.3)
\item Required content: one or more model kinds (per 8.2) implemented by the ADL for use in framing the relevant concerns or reflect the relevant aspects
\item Required content: any architecture viewpoints (per 8.1) implemented by the ADL
\item Required content: any correspondence methods (per 6.9.3)
\item Required content: version identification as specified by the organization and/or project
\end{itemize}}
\CornellNoteRow{Related sections}{6.4, 6.5, 8.3, 8.2, 8.1, 6.9.3}
\CornellNoteRow{Conformance check}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Required content: identification of typical concerns (6.4) or aspects (6.5) covered by the ADL
\item Required content: the identification of one or more view methods to be selected from the ADL (per 8.3)
\end{itemize}}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Which provisions in Specification of an architecture description language are mandatory for conformance?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
Defines the minimum specification content for an architecture description language. For conformance, its mandatory provisions must be satisfied, including: Required content: identification of typical concerns (6.4) or aspects (6.5) covered by the ADL; Required content: the identification of one or more view methods to be selected from the ADL (per 8.3).
\end{CornellSummaryBox}
