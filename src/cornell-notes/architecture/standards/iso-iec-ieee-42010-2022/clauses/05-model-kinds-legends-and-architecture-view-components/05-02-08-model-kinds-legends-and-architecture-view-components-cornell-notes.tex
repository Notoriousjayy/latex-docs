\begin{CornellOverviewBox}{Overview}
{\Large\bfseries\textcolor{CNavy}{Section 5.2.8: Model kinds, legends and architecture view components}}\\[3pt]
{\small\textbf{Classification:} Conceptual foundation}\\
{\small\textbf{Learning objective:} Explain the role of model kinds, legends and architecture view components and connect it to related architecture-description concepts.}
\end{CornellOverviewBox}
\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Purpose / key concept}{An architecture view is composed of one or more architecture view components.}
\CornellNoteRow{Core details}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item A view component that can be based on a model or not.
\item Each view component is governed by a model kind or legend identified by its architecture viewpoint.
\item A model kind determines the conventions for model-based view components.
\item A legend documents the conventions for view components.
\end{itemize}}
\CornellNoteRow{Distinction}{A view component is part of one or more views; a model kind supplies conventions for model-based components, while a legend documents conventions where no model kind governs.}
\CornellNoteRow{Study relationship / visual insight}{Study relationship: a viewpoint can specify model kinds and governs views; views are composed of view components; model kinds determine conventions for model-based components, while non-model-based components rely on documented legends.}
\CornellNoteRow{Example / application}{Model kinds include use cases, activity models [such as structured analysis and design technique (SADT) and ICAM1) Definition Language (IDEF0)], threat models, component models and connectivity models.}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item What is the central role of Model kinds, legends and architecture view components?
\item How does Model kinds, legends and architecture view components connect to adjacent architecture-description concepts?
\item What closely related concepts must be kept distinct?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
An architecture view is composed of one or more architecture view components. A view component is part of one or more views; a model kind supplies conventions for model-based components, while a legend documents conventions where no model kind governs.
\end{CornellSummaryBox}
