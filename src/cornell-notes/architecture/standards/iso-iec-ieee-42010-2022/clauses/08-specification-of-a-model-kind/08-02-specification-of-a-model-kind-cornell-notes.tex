\begin{CornellOverviewBox}{Overview}
{\Large\bfseries\textcolor{CNavy}{Section 8.2: Specification of a model kind}}\\[3pt]
{\small\textbf{Classification:} Normative}\\
{\small\textbf{Learning objective:} Identify the mandatory, recommended, and permitted provisions governing specification of a model kind.}
\end{CornellOverviewBox}
\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Purpose}{Defines the minimum content of a model-kind specification.}
\CornellNoteRow{Requirement}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item The specification of a model kind must include or reference:
\item Required content: conventions such as definition of a language, notation or modelling technique comprising the model kind
\item Required content: any view methods (per 8.3) and correspondence methods associated with the model kind
\item Required content: any version identification as specified by the organization and/or project
\item Required content: any sources of information about this model kind
\end{itemize}}
\CornellNoteRow{Interpretive note}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Item a) can be met in a number of ways such as with a metamodel, grammar or template for the specification of a model kind that defines the structure and interpretation of its models (see B.2.9).
\end{itemize}}
\CornellNoteRow{Related sections}{8.3, B.2.9}
\CornellNoteRow{Conformance check}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Required content: conventions such as definition of a language, notation or modelling technique comprising the model kind
\item Required content: any view methods (per 8.3) and correspondence methods associated with the model kind
\end{itemize}}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Which provisions in Specification of a model kind are mandatory for conformance?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
Defines the minimum content of a model-kind specification. For conformance, its mandatory provisions must be satisfied, including: Required content: conventions such as definition of a language, notation or modelling technique comprising the model kind; Required content: any view methods (per 8.3) and correspondence methods associated with the model kind.
\end{CornellSummaryBox}
