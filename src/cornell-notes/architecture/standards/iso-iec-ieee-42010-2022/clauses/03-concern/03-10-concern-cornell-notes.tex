\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Term}{\textbf{concern}}
\CornellNoteRow{Definition}{A matter that is relevant or important to a stakeholder.}
\CornellNoteRow{Key characteristics}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item concerns can be identified with regards to an entity of interest (3.12) or independently, such as with regards to environment, scenario, situation or use case of that entity.
\item Here, interest in an entity is intended to encompass interest in that entity’s environment (3.13), life cycle, architecture (3.2), requirements, design, implementation and operation. Such interests are captured via aspects (3.9), concerns and stakeholder perspectives (3.18).
\item The identification of a concern is often the result of prior knowledge, experience and praxis in the domain to which the concern applies.
\item See 5.2.3 for more discussion and examples.
\end{itemize}}
\CornellNoteRow{Distinction}{A concern belongs to stakeholder interests; it can be refined or examined through aspects and framed by viewpoints.}
\CornellNoteRow{Related sections}{3.17, 3.12, 3.13, 3.2, 3.9, 3.18, 5.2.3, 3.8}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item How would you explain concern in your own words?
\item Which neighboring concept is easiest to confuse with concern, and how is it different?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
A matter that is relevant or important to a stakeholder. A concern belongs to stakeholder interests; it can be refined or examined through aspects and framed by viewpoints.
\end{CornellSummaryBox}
