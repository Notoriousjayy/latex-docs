\begin{CornellNotesTable}
\CornellNoteRow{Term}{\textbf{architecture description}}
\CornellNoteRow{Definition}{A tangible work product used to express an architecture.}
\CornellNoteRow{Key characteristics}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item A work product is an artifact produced by a process (see ISO/IEC 20246:2017, 3.18).
\item An architecture description is a tangible representation of information provided to the stakeholders (3.17). An AD is considered an information part (3.14).
\end{itemize}}
\CornellNoteRow{Distinction}{Do not equate the description with the architecture itself: the description is a work product, while the architecture is the underlying concepts/properties and governing principles.}
\CornellNoteRow{Related sections}{3.2, 3.18, 3.17, 3.14}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item How would you explain architecture description in your own words?
\item Which neighboring concept is easiest to confuse with architecture description, and how is it different?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
A tangible work product used to express an architecture. Do not equate the description with the architecture itself: the description is a work product, while the architecture is the underlying concepts/properties and governing principles.
\end{CornellSummaryBox}
