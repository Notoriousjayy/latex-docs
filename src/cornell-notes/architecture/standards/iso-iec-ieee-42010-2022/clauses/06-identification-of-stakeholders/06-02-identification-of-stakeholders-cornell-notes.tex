\begin{CornellOverviewBox}{Overview}
{\Large\bfseries\textcolor{CNavy}{Section 6.2: Identification of stakeholders}}\\[3pt]
{\small\textbf{Classification:} Normative}\\
{\small\textbf{Learning objective:} Identify the mandatory, recommended, and permitted provisions governing identification of stakeholders.}
\end{CornellOverviewBox}
\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Purpose}{Establishes how fundamental stakeholders, known impacts, limitations, and non-conformances are addressed in an architecture description.}
\CornellNoteRow{Requirement}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item A conforming architecture description must identify the stakeholders having concerns that are considered fundamental to the architecture of the entity of interest and consistent with the purpose of the AD.
\item Consideration must be given to recommendations made in previous evaluations of the architecture or of related architectures.
\item A conforming architecture description must include a statement of known resource limitations or other constraints that prevented the AD from addressing identified stakeholders and architecture considerations, i.e. concerns, perspectives, and aspects.
\item Non-conformances must be identified and explained with rationales.
\end{itemize}}
\CornellNoteRow{Recommendation}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item An architecture description should identify the possible impacts of the architecture on the current and future stakeholders.
\end{itemize}}
\CornellNoteRow{Example / application}{Stakeholders include users, operators, acquirers, owners, suppliers and vendors, architects, designers and developers, implementers, maintainers, regulators (including government), testers, public-atlarge, adversaries and competitors.}
\CornellNoteRow{Conformance check}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item A conforming architecture description must identify the stakeholders having concerns that are considered fundamental to the architecture of the entity of interest and consistent with the purpose of the AD
\item Consideration must be given to recommendations made in previous evaluations of the architecture or of related architectures
\end{itemize}}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Which provisions in Identification of stakeholders are mandatory for conformance?
\item Which provisions are recommendations or permissions rather than requirements?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
A conforming architecture description must identify stakeholders whose concerns are fundamental, consider recommendations from prior architecture evaluations, record relevant resource or constraint limitations, and explain non-conformances with rationale. It should also identify possible impacts on current and future stakeholders.
\end{CornellSummaryBox}
