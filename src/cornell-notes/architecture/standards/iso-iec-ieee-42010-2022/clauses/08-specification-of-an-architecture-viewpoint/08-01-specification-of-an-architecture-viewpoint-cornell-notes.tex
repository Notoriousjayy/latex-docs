\begin{CornellOverviewBox}{Overview}
{\Large\bfseries\textcolor{CNavy}{Section 8.1: Specification of an architecture viewpoint}}\\[3pt]
{\small\textbf{Classification:} Normative}\\
{\small\textbf{Learning objective:} Identify the mandatory, recommended, and permitted provisions governing specification of an architecture viewpoint.}
\end{CornellOverviewBox}
\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Purpose}{Defines the minimum content of an architecture viewpoint specification and its relationships to concerns, aspects, stakeholders, model kinds, legends, and correspondence methods.}
\CornellNoteRow{Core details}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item The specification of an architecture viewpoint can be included as part of an AD (Clause 6), as a part of the specification of an ADF or ADL (Clause 7) or individually using the requirements of this clause.
\end{itemize}}
\CornellNoteRow{Requirement}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item The specification of an architecture viewpoint must include or reference:
\item Required content: any stakeholder perspectives associated with this viewpoint (per 6.3)
\item Required content: one or more concerns (per 6.4) framed by this architecture viewpoint
\item Required content: one or more aspects related to those concerns (per 6.5)
\item Required content: known typical stakeholders (per 6.2) holding those concerns which are framed by this architecture viewpoint (per item b))
\item Required content: model kinds (per 8.2) and legends for use when constructing views (per 8.2)
\item Required content: correspondence methods capturing relations within resulting views and their view components (per 6.8)
\item Required content: references to any sources of information about this viewpoint. A specification of an architecture viewpoint should identify view methods (per 8.3) used to create, interpret or analyse views governed by the associated architecture viewpoint; and one or more model kind (per 8.2). Each legend must provide guidance to users in interpreting the view components which it documents. The specification of an architecture viewpoint can use correspondence methods
\end{itemize}}
\CornellNoteRow{Interpretive note}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item When a specification of an architecture viewpoint is included and applied in an AD, the typical stakeholders of item e) are replaced by the known stakeholders identified in the AD.
\item This section does not require any particular specifications of architecture viewpoints to be used.
\end{itemize}}
\CornellNoteRow{Related sections}{6.3, 6.4, 6.5, 6.2, 8.2, 6.8, 8.3}
\CornellNoteRow{Conformance check}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Required content: any stakeholder perspectives associated with this viewpoint (per 6.3)
\item Required content: one or more concerns (per 6.4) framed by this architecture viewpoint
\end{itemize}}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item Which provisions in Specification of an architecture viewpoint are mandatory for conformance?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
Defines the minimum content of an architecture viewpoint specification and its relationships to concerns, aspects, stakeholders, model kinds, legends, and correspondence methods. For conformance, its mandatory provisions must be satisfied, including: Required content: any stakeholder perspectives associated with this viewpoint (per 6.3); Required content: one or more concerns (per 6.4) framed by this architecture viewpoint.
\end{CornellSummaryBox}
