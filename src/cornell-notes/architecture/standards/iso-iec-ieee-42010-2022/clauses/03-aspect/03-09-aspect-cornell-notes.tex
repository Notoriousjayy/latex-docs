\vspace{3pt}

\begin{CornellNotesTable}
\CornellNoteRow{Term}{\textbf{aspect}}
\CornellNoteRow{Definition}{A part of an entity’s character or nature.}
\CornellNoteRow{Key characteristics}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item A particular aspect can be used for capturing the relevant features of the entity of interest (3.12) as a refinement of one or more concerns (3.10) under examination with respect to some part of its character, e.g. the structural character, functional character or informational character of the entity.
\item Aspects enable the architect to analyse, address and structure concerns (3.10). In general, there is a many-to-many relation between aspects and concerns (3.10).
\item See 5.2.5 for more discussion and examples.
\end{itemize}}
\CornellNoteRow{Distinction}{An aspect is part of the entity’s character used to structure analysis; a concern is a matter of stakeholder relevance or importance.}
\CornellNoteRow{Example / application}{Functional, structural and informational aspects of an entity.}
\CornellNoteRow{Related sections}{3.12, 3.10, 5.2.5}
\CornellNoteRow{Active recall}{\begin{itemize}[leftmargin=*,nosep,topsep=1pt]
\item How would you explain aspect in your own words?
\item Which neighboring concept is easiest to confuse with aspect, and how is it different?
\end{itemize}}
\end{CornellNotesTable}


\begin{CornellSummaryBox}{Summary}
A part of an entity’s character or nature. An aspect is part of the entity’s character used to structure analysis; a concern is a matter of stakeholder relevance or importance.
\end{CornellSummaryBox}
